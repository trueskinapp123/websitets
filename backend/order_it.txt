from collections import deque

def get_neighbors(arr):
    n = len(arr)
    for i in range(n):
        for j in range(i, n):
            seg = arr[i:j+1]
            rem = arr[:i] + arr[j+1:]
            for k in range(len(rem) + 1):
                if k == i:
                    continue
                new = rem[:k] + seg + rem[k:]
                if new != arr:
                    yield new

n = int(input().strip())
input()
shuffled = [input().rstrip('\n') for _ in range(n)]
input()
original = [input().rstrip('\n') for _ in range(n)]

start = tuple(shuffled)
target = tuple(original)

if start == target:
    print(0, end='')
    exit()

front = {start}
back = {target}
seen_front = {start: 0}
seen_back = {target: 0}

while front and back:
    if len(front) > len(back):
        front, back = back, front
        seen_front, seen_back = seen_back, seen_front

    next_front = set()
    for arr in front:
        d = seen_front[arr]
        for new in get_neighbors(arr):
            if new in seen_front:
                continue
            if new in seen_back:
                print(d + 1 + seen_back[new], end='')
                exit()
            seen_front[new] = d + 1
            next_front.add(new)
    front = next_front

print(-1, end='')
